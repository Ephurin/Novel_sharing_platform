# 1) Deterministic recent-only baseline (no API key needed)

  --output ../docs/development/P7-01-EMBEDDING-BENCHMARK-ADR/baseline-run.json

# Requires GEMINI_API_KEY or GOOGLE_API_KEY in the local environment.

  --provider gemini \
  --model gemini-embedding-001 \
  --dimension 768 \
  --batch-size 32 \
  --output ../docs/development/P7-01-EMBEDDING-BENCHMARK-ADR/gemini-001-768.json
